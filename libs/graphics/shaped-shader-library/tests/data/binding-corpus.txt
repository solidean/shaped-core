# The binding pass's shared corpus: HLSL snippets and the parse each one must produce.
#
# The pass exists twice -- once in C++ for the runtime rewriter, once in Python for the build-time generator --
# and this file is what keeps the two agreeing on a grammar case rather than on two separate readings of it.
# So a case is added here once, and both sides pick it up.
#
# It does not prove the two parsers are the same function. It proves they agree on what we thought of; the
# per-package self-check covers what we ship, and the reflection cross-check is the only leg that asks DXC.
#
# Format, line-oriented, with every marker at column 0:
#
#   --- case <what this pins>
#   --- hlsl            the snippet, verbatim, its first line being line 1 of every error message below
#   --- groups          the expected parse, or nothing at all when the snippet declares no group
#   <namespace> group=<n>
#     <name> index=<i> type=<sg::binding_type> count=<n> [dim=<sg::texture_view_dimension>]
#   --- error           instead of `groups`: the message the pass must report, exactly
#
# A `#` comment is only a comment outside an `--- hlsl` block; inside one it is HLSL text like any other.


--- case a texture and a sampler share one counter, across register classes
--- hlsl
namespace frame_bindings //!> group 0
{
    Texture2D<float4> albedo;
    SamplerState linear_sampler;
}
--- groups
frame_bindings group=0
  albedo index=0 type=readonly_texture count=1 dim=tex_2d
  linear_sampler index=1 type=sampler count=1


--- case two groups, each numbered by its own marker rather than by order
--- hlsl
namespace frame //!> group 0
{
    ConstantBuffer<frame_constants> constants;
}

namespace material //!> group 2
{
    Texture2D<float4> base_color;
    RWStructuredBuffer<uint> counters;
}
--- groups
frame group=0
  constants index=0 type=uniform_buffer count=1
material group=2
  base_color index=0 type=readonly_texture count=1 dim=tex_2d
  counters index=1 type=readwrite_structured_buffer count=1

  t2dmsa index=5 type=readonly_texture count=1 dim=tex_2d_ms_array
  t3d index=6 type=readonly_texture count=1 dim=tex_3d
  tcube index=7 type=readonly_texture count=1 dim=cube
